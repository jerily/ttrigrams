cmake_minimum_required(VERSION 3.18)
project(ttrigrams VERSION 0.2.0 LANGUAGES C)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(TTRIGRAMS_BUILD_TCL "Build the Tcl extension" ON)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# The C library is always static and has no Tcl dependency.
add_library(ttrigrams_core STATIC src/ttrigrams.c)
add_library(ttrigrams::core ALIAS ttrigrams_core)
set_target_properties(ttrigrams_core PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME core)
target_include_directories(ttrigrams_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/ttrigrams>)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ttrigrams_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ttrigrams_core EXPORT ttrigramsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/ttrigrams.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/ttrigrams)
install(EXPORT ttrigramsTargets NAMESPACE ttrigrams:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrigrams)
configure_package_config_file(cmake/ttrigramsConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ttrigramsConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrigrams)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ttrigramsConfigVersion.cmake
    COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ttrigramsConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/ttrigramsConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrigrams)

if(TTRIGRAMS_BUILD_TCL)
    find_path(TCL_INCLUDE_PATH NAMES tcl.h HINTS ${TCL_INCLUDE_DIR}
        PATH_SUFFIXES tcl9.1 tcl9.0 tcl8.6 tcl REQUIRED)
    file(STRINGS "${TCL_INCLUDE_PATH}/tcl.h" TCL_VERSION_LINE REGEX "^#[ \t]*define TCL_VERSION[ \t]+\"")
    string(REGEX MATCH "[0-9]+\\.[0-9]+" TTRIGRAMS_TCL_VERSION "${TCL_VERSION_LINE}")
    if(TTRIGRAMS_TCL_VERSION VERSION_LESS 8.6 OR NOT TTRIGRAMS_TCL_VERSION VERSION_LESS 10)
        message(FATAL_ERROR "Tcl 8.6 or 9.x is required")
    endif()
    if(TTRIGRAMS_TCL_VERSION VERSION_LESS 9)
        set(TTRIGRAMS_TCL_REQUIREMENT "8.6-9")
    else()
        set(TTRIGRAMS_TCL_REQUIREMENT "9.0-10")
    endif()
    find_program(TCL_TCLSH NAMES tclsh${TTRIGRAMS_TCL_VERSION} HINTS "${TCL_INCLUDE_PATH}/../bin")
    add_library(ttrigrams src/ttrigrams_tcl.c)
    target_include_directories(ttrigrams PRIVATE ${TCL_INCLUDE_PATH})
    target_compile_definitions(ttrigrams PRIVATE TTRIGRAMS_VERSION="${PROJECT_VERSION}")
    target_link_libraries(ttrigrams PRIVATE ttrigrams_core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ttrigrams PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(BUILD_SHARED_LIBS)
        string(REPLACE "." "" TTRIGRAMS_TCL_COMPACT_VERSION "${TTRIGRAMS_TCL_VERSION}")
        find_library(TCL_STUB_LIBRARY NAMES tclstub${TTRIGRAMS_TCL_VERSION} tclstub${TTRIGRAMS_TCL_COMPACT_VERSION} tclstub
            HINTS ${TCL_LIBRARY_DIR} "${TCL_INCLUDE_PATH}/../lib" REQUIRED)
        target_compile_definitions(ttrigrams PRIVATE USE_TCL_STUBS)
        target_link_libraries(ttrigrams PRIVATE ${TCL_STUB_LIBRARY})
        set(TTRIGRAMS_LOAD "[list load [file join $dir ${CMAKE_SHARED_LIBRARY_PREFIX}ttrigrams${CMAKE_SHARED_LIBRARY_SUFFIX}] Ttrigrams]")
    else()
        target_compile_definitions(ttrigrams PRIVATE STATIC_BUILD)
        set(TTRIGRAMS_LOAD "[list load {} Ttrigrams]")
    endif()
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
    install(TARGETS ttrigrams LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/ttrigrams${PROJECT_VERSION}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl DESTINATION ${CMAKE_INSTALL_LIBDIR}/ttrigrams${PROJECT_VERSION})
endif()

if(BUILD_TESTING)
    add_executable(test_core tests/core.c)
    target_link_libraries(test_core PRIVATE ttrigrams_core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_core PRIVATE -UNDEBUG)
    endif()
    add_test(NAME core COMMAND test_core)
    if(TTRIGRAMS_BUILD_TCL)
        if(BUILD_SHARED_LIBS)
            if(NOT TCL_TCLSH)
                message(FATAL_ERROR "Tcl shell is required for tests")
            endif()
            set(TTRIGRAMS_TEST_SHELL "${TCL_TCLSH}")
            if(UNIX AND CMAKE_NM)
                add_test(NAME tcl_stubs COMMAND ${CMAKE_COMMAND}
                    -DNM=${CMAKE_NM} -DMODULE=$<TARGET_FILE:ttrigrams>
                    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/stubs.cmake)
            endif()
        else()
            # Only the embedding test executable needs the full Tcl runtime.
            find_library(TCL_LIBRARY NAMES tcl${TTRIGRAMS_TCL_VERSION}
                HINTS ${TCL_LIBRARY_DIR} "${TCL_INCLUDE_PATH}/../lib" REQUIRED)
            add_executable(test_embed tests/embed.c)
            target_include_directories(test_embed PRIVATE ${TCL_INCLUDE_PATH})
            target_link_libraries(test_embed PRIVATE ttrigrams ttrigrams_core ${TCL_LIBRARY})
            set(TTRIGRAMS_TEST_SHELL "$<TARGET_FILE:test_embed>")
        endif()
        add_test(NAME tcl COMMAND ${TTRIGRAMS_TEST_SHELL} ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl)
        set_tests_properties(tcl PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()
